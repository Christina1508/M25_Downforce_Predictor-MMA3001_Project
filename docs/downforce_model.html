downforce_model

Numerical prediction of M25 full-vehicle downforce area.

Fits linear, quadratic and Gaussian RBF regression models using wind tunnel airspeed and yaw measurements. Performs cross-validation, hyperparameter optimisation, independent validation and aerodynamic sensitivity analysis.

Inputs: Airspeed (m/s) and yaw angle (deg). Output: Predicted downforce area, CLA (m^2).

Run after data/data_extract.py to generate the required CSV dataset.

180# Gaussian RBF kernel function
181def rbf_kernel(X1, X2):
182    """Calculate the Gaussian RBF kernel using standardised inputs.
183
184    Args:
185        X1 (np.ndarray): First set of airspeed and yaw inputs.
186        X2 (np.ndarray): Second set of airspeed and yaw inputs.
187
188    Returns:
189        np.ndarray: Gaussian kernel matrix, shape (len(X1), len(X2)).
190    """
191    X1_scaled = (X1 - x_mean) / x_std
192    X2_scaled = (X2 - x_mean) / x_std
193
194    # Squared distances between all pairs of points
195    dist_sq = np.sum(
196        (X1_scaled[:, None, :] - X2_scaled[None, :, :])**2,
197        axis=2
198    )
199
200    return np.exp(-dist_sq / (2 * length_scale**2))

308# Fit a model and predict unseen inputs
309def fit_predict(model, X_fit, y_fit, X_test):
310    """Fit a regression model and predict withheld measurements.
311
312    Used during cross-validation to train on the selected subset without
313    including withheld measurements.
314
315    Args:
316        model (str): "Linear", "Quadratic", or "Gaussian RBF".
317        X_fit (np.ndarray): Training airspeed and yaw inputs.
318        y_fit (np.ndarray): Training CLA values (m^2).
319        X_test (np.ndarray): Inputs at which predictions are required.
320
321    Returns:
322        np.ndarray: Predicted CLA values (m^2), or None for an
323            unsupported model name.
324    """
325
326    if model == "Linear":
327        A_fit = np.column_stack((np.ones(len(X_fit)), X_fit))
328        A_test = np.column_stack((np.ones(len(X_test)), X_test))
329
330        coeff, _, _, _ = np.linalg.lstsq(A_fit, y_fit, rcond=None)
331        return A_test @ coeff
332
333    elif model == "Quadratic":
334        A_fit = quadratic_features(X_fit)
335        A_test = quadratic_features(X_test)
336
337        coeff, _, _, _ = np.linalg.lstsq(A_fit, y_fit, rcond=None)
338        return A_test @ coeff
339
340    elif model == "Gaussian RBF":
341        # Standardise using ONLY the current training fold
342        mean = X_fit.mean(axis=0)
343        std = X_fit.std(axis=0)
344        std = np.where(std == 0, 1, std)
345
346        def kernel(X1, X2):
347            """Calculate the Gaussian kernel using training-fold scaling."""
348            X1 = (X1 - mean) / std
349            X2 = (X2 - mean) / std
350
351            dist_sq = np.sum(
352                (X1[:, None, :] - X2[None, :, :])**2,
353                axis=2
354            )
355            return np.exp(-dist_sq / (2 * length_scale**2))
356
357        K = kernel(X_fit, X_fit)
358        y_mean_fold = y_fit.mean()
359
360        weights = np.linalg.solve(
361            K + regularisation * np.eye(len(X_fit)),
362            y_fit - y_mean_fold
363        )
364
365        return y_mean_fold + kernel(X_test, X_fit) @ weights

542# Function to fit and predict hyperparams
543def rbf_cv_predict(X_fit, y_fit, X_test, ls, reg):
544    """Fit and evaluate a Gaussian RBF model with specified parameters.
545
546    Standardises inputs using the training fold to avoid data leakage
547    during hyperparameter optimisation.
548
549    Args:
550        X_fit (np.ndarray): Training airspeed and yaw inputs.
551        y_fit (np.ndarray): Training CLA values (m^2).
552        X_test (np.ndarray): Withheld inputs for prediction.
553        ls (float): Gaussian kernel length scale.
554        reg (float): Regularisation parameter.
555
556    Returns:
557        np.ndarray: Predicted CLA values (m^2).
558    """
559
560    # Standardise inputs using training fold only
561    mean = X_fit.mean(axis=0)
562    std = X_fit.std(axis=0)
563    std = np.where(std == 0, 1, std)
564
565    def kernel(X1, X2):
566        """Calculate the Gaussian kernel using the specified length scale."""
567        X1 = (X1 - mean) / std
568        X2 = (X2 - mean) / std
569
570        dist_sq = np.sum(
571            (X1[:, None, :] - X2[None, :, :])**2,
572            axis=2
573        )
574
575        return np.exp(-dist_sq / (2 * ls**2))
576
577    # Fit RBF model
578    K = kernel(X_fit, X_fit)
579    y_mean_fold = y_fit.mean()
580
581    weights = np.linalg.solve(
582        K + reg * np.eye(len(X_fit)),
583        y_fit - y_mean_fold
584    )
585
586    # Predict withheld measurements
587    return y_mean_fold + kernel(X_test, X_fit) @ weights
